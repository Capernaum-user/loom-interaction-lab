앱 셸프로그램 창3단 레이아웃타이틀바 본문 상태줄

수치 6개
제목줄 높이44px
상태줄 높이26px
레일 폭264px접으면 58px. 이 변수 하나로만 바꾼다
창 높이100dvh100vh 면 주소창이 접힐 때 상태줄이 잘린다
본문 열minmax(0,1fr)1fr 로 적으면 긴 표 한 줄이 레일을 밀어낸다
본문 여백14px 16px 40px
소스 18줄 · grid-template-areas, minmax(), dvh, CSS 변수
/* 3띠 골격 — 제목줄 44px · 본문 1fr · 상태줄 26px. 어떤 조작에도 창 뼈대가 흔들리지 않는다. */
#app{display:grid;height:100dvh;                          /* 100vh 가 아니다 — 주소창이 접혀도 상태줄이 안 잘린다 */
  grid-template-columns:var(--rail,264px) minmax(0,1fr);  /* 1fr 로 적으면 긴 표 한 줄이 레일을 화면 밖으로 민다 */
  grid-template-rows:44px minmax(0,1fr) 26px;
  grid-template-areas:"title title" "rail main" "status status"}
#app[data-rail="off"]{--rail:58px}                        /* 레일 폭은 이 변수 하나로만 바꾼다 */
#titlebar{grid-area:title;display:flex;align-items:center;gap:10px;padding:0 10px;
  background:var(--surface);border-bottom:1px solid var(--line);position:relative;z-index:20}
#rail{grid-area:rail;overflow-y:auto;overflow-x:hidden;
  background:var(--surface-2);border-right:1px solid var(--line)}
#main{grid-area:main;overflow:auto;position:relative;padding:14px 16px 40px}
#statusbar{grid-area:status;display:flex;align-items:center;gap:14px;padding:0 12px;
  background:var(--surface);border-top:1px solid var(--line)}
body{margin:0;overflow:hidden}  /* 스크롤은 #main·#rail 이 각자 갖는다 — body 가 구르면 상태줄이 밀려난다 */

const TARGET = '관제 화면 여덟 가지 · 외부 라이브러리 0 · 표준 API 만 · READY';
const STEP = 42, EVERY = 3;
/* 글자 폭이 다른 것을 섞으면 해독되는 동안 줄이 흔들린다.
   한글 자리는 한글로, 로마자 자리는 로마자로 뒤섞어 폭을 고정한다. */
const POOL_EN = 'ABCDEFGHIJKLMNOPQRSTUVWXYZ0123456789#%&*+=<>/';
const POOL_KO = '가나다라마바사아자차카타파하거너더러머버서어저처커터퍼허고노도로모보소오조초코토포호';
const pick = ch => {
  if (ch === ' ') return ' ';
  const c = ch.charCodeAt(0);
  if (c >= 0xAC00 && c <= 0xD7A3) return POOL_KO[Math.random() * POOL_KO.length | 0];
  if (c > 0x2000) return ch;                       /* 가운뎃점 같은 기호는 그대로 둔다 */
  return POOL_EN[Math.random() * POOL_EN.length | 0];
};

const dec = document.getElementById('dec');
dec.innerHTML = '<span data-done></span><span data-scr class="scr"></span>';
const A = dec.querySelector('[data-done]'), B = dec.querySelector('[data-scr]');
let t0 = 0, n = 0, done = false;

/* 화면에 들어오기 전에도 뒤섞인 채로 보인다 — 빈 줄이 놓여 있으면 고장으로 읽힌다.
   제자리를 찾기 «시작하는 시점» 만 관찰자가 정한다. */
new IntersectionObserver((es, ob) => es.forEach(e => {
  if (e.isIntersecting) { t0 = performance.now(); ob.unobserve(e.target); }
}), { threshold: .4 }).observe(dec);

function frame(now) {
  requestAnimationFrame(frame);
  if (done) return;
  const settled = t0 ? Math.floor((now - t0) / STEP) : 0;
  if (t0 && settled >= TARGET.length) { dec.textContent = TARGET; done = true; return; }
  if (++n % EVERY) return;                         /* 매 프레임 바꾸면 글자가 읽히지 않는다 */
  let s = '';
  for (let i = settled; i < TARGET.length; i++) s += pick(TARGET[i]);
  A.textContent = TARGET.slice(0, settled);
  B.textContent = s;
}
requestAnimationFrame(frame);
</script>
